\subsubsection{Berechnung}

\begin{align}
	Z_E              & = jX_k \parallel \bigl(jX_{AC} + (jX_{E} \parallel R_E)\bigr) \\
	%                  & = \frac{ 1 + \mathrm{j}\,\omega\!\bigl(R_E C_{E} + R_E C_{AC}\bigr)}
	% {\omega^2 R_E\!\bigl(-C_E C_k - C_{AC} C_k - C_E C_{AC}\bigr)
	% + \mathrm{j}\,\omega\!\bigl(C_k + C_{AC}\bigr)}                                         \\[4pt]
	\lvert Z_E\rvert & =
	\frac{\sqrt{\,1 + \bigl(\omega R_E (C_{E}+C_{AC})\bigr)^2\,}}
	{\omega\,\sqrt{\,\bigl(\omega R_E (C_E C_k + C_{AC} C_k + C_E C_{AC})\bigr)^2
	+ (C_k + C_{AC})^2\,}}                                                           \\[4pt]
	                 & \approx \SI{1326.5}{\ohm}. \label{eq:eingangswiderstand_oszi}
\end{align}
