\marginref{\ref{itm:1-P2-2},~\ref{itm:1-P2-3}}
Die Berechnung des Mittelwertes $\bar{R}$ und der Standardabweichung~$s$~\cite[S.~7]{EMTPraktikum2025}:
\begin{align}
	 & \bar{R} = \frac{1}{n} \sum_{i=1}^{n} R_i = \SI{67.94}{\ohm}                                            \\
	 & s = \sqrt{\frac{1}{n-1} \sum_{i=1}^n {(R_i - \bar{R})}^2} = \SI{0.08451}{\ohm} \approx \SI{0.08}{\ohm}
\end{align}

Die zufällige Messunsicherheit berechnet sich nach~\cite[S.~9, Gl.~1.13]{EMTPraktikum2025} zu
\[
	u_{z} = t_{0} \frac{s}{\sqrt{n}} \, .
\]

Für $n=5$ ergibt sich:

\begin{itemize}
	\item 68,26\% Vertrauensniveau ($t_{0}/\sqrt{n} = 0.51$):
	      \[
		      \begin{gathered}
			      u_{z} = 0.51 \cdot \SI{0.08}{\ohm} \approx \SI{0.041}{\ohm} \\
			      R = \bar{R} \pm u_{z} = (67.94 \pm 0.04) \, \si{\ohm}
		      \end{gathered}
	      \]
	      also das Konfidenzintervall $[67.90; 67.98]\si{\ohm}$
	      also das Konfidenzintervall $[67.90; 67.98]\si{\ohm}$

	\item 99,5\% Vertrauensniveau ($t_{0}/\sqrt{n} = 2.50$):
	      \[
		      \begin{gathered}
			      u_{z} = 2.50 \cdot \SI{0.08}{\ohm} \approx \SI{0.175}{\ohm} \\
			      R = \bar{R} \pm u_{z} = (67.94 \pm 0.18) \, \si{\ohm}
		      \end{gathered}
	      \]
	      also das Konfidenzintervall $[67.76; 68.12]\si{\ohm}$
	      also das Konfidenzintervall $[67.76; 68.12]\si{\ohm}$
\end{itemize}
